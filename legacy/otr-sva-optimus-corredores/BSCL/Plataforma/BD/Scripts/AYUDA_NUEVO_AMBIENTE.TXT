 2) Configuracion de permisos en base datos HANA
----------------------------------------------------------------------------
 
	CREATE SCHEMA BSCL;
	CREATE SCHEMA RUEBAS; 
	
	CREATE ROLE BCS; 
	
	GRANT SELECT ON SCHEMA "BSCL" TO BCS;
	GRANT INSERT, UPDATE, DELETE, EXECUTE ON SCHEMA "BSCL" TO BCS;
	GRANT CREATE ANY ON SCHEMA "BSCL" TO BCS;
	GRANT SESSION ADMIN TO BCS; 

	
	GRANT SELECT ON SCHEMA "RUEBAS" TO BCS;
	GRANT INSERT, UPDATE, DELETE, EXECUTE ON SCHEMA "RUEBAS" TO BCS;
	GRANT CREATE ANY ON SCHEMA "RUEBAS" TO BCS; 
	GRANT ALTER,DROP ON SCHEMA "BSCL" TO BCS;
  
	GRANT EXECUTE ON "BSCL" to BCS;
	GRANT REPO.READ ON _SYS_REPO."BSCL" TO BCS;
	GRANT REPO.EDIT_NATIVE_OBJECTS ON _SYS_REPO."BSCL" TO BCS;
	GRANT REPO.ACTIVATE_NATIVE_OBJECTS ON _SYS_REPO."BSCL" TO BCS;
	GRANT REPO.MAINTAIN_NATIVE_PACKAGES ON _SYS_REPO."BSCL" TO BCS;
	GRANT REPO.ACTIVATE_IMPORTED_OBJECTS ON _SYS_REPO."BSCL" TO BCS;
	GRANT REPO.MAINTAIN_IMPORTED_PACKAGES ON _SYS_REPO."BSCL" TO BCS;	
	GRANT REPO.EDIT_IMPORTED_OBJECTS ON _SYS_REPO."BSCL" TO BCS;
 
	GRANT SELECT ON SCHEMA "_SYS_BI" TO BCS; 
	GRANT SELECT ON SCHEMA "_SYS_BIC" TO BCS; 
	GRANT EXECUTE ON SCHEMA "_SYS_BI" TO BCS; 
	GRANT EXECUTE ON SCHEMA "_SYS_BIC" TO BCS; 
	GRANT EXECUTE ON SCHEMA "BSCL" to _SYS_REPO;
	
	GRANT EXECUTE,SELECT,INSERT,UPDATE,DELETE ON SCHEMA "_SYS_REPO" TO BCS; 
	GRANT EXECUTE,SELECT,INSERT,UPDATE,DELETE ON SCHEMA "RUEBAS" TO BCS; 
	GRANT EXECUTE,SELECT,INSERT,UPDATE,DELETE ON SCHEMA "BSCL_VIRTUAL" TO BCS;
